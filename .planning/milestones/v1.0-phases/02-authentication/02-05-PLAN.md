  ```csharp
  // SPDX-License-Identifier: GPL-3.0-or-later
  // Copyright (c) 2026 GameKit contributors

  using System;
  using System.Collections.Generic;
  using System.Net.Http;
  using System.Threading;
  using System.Threading.Tasks;
  using Microsoft.AspNetCore.Http;

  namespace GameKit.Auth.Providers.Steam;

  /// &lt;summary&gt;
  /// In-house Steam OpenID 2.0 verifier (CONTEXT D-09). POSTs &lt;c&gt;check_authentication&lt;/c&gt; to the
  /// OP endpoint using the named HttpClient &lt;c&gt;gamekit.auth.provider.steam&lt;/c&gt; (which has the
  /// &lt;see cref="Egress.EgressAllowListHandler"/&gt; + resilience pipeline attached). See
  /// &lt;see href="https://openid.net/specs/openid-authentication-2_0.html#verifying_signatures"/&gt;.
  /// &lt;/summary&gt;
  public sealed class SteamOpenIdVerifier
  {
      private readonly HttpClient _httpClient;
      private readonly GameKitAuthOptions _opts;

      /// &lt;summary&gt;Constructs the verifier; pulls the pre-configured HttpClient from the factory.&lt;/summary&gt;
      public SteamOpenIdVerifier(IHttpClientFactory factory, GameKitAuthOptions opts)
      {
          _httpClient = factory.CreateClient("gamekit.auth.provider.steam");
          _opts = opts;
      }

      /// &lt;summary&gt;
      /// Verifies an OpenID 2.0 positive assertion by roundtripping &lt;c&gt;check_authentication&lt;/c&gt;.
      /// Returns a SteamID64 on success; an error code on failure.
      /// &lt;/summary&gt;
      public async Task&lt;SteamVerificationResult&gt; VerifyAsync(IQueryCollection query, CancellationToken cancellationToken = default)
      {
          ArgumentNullException.ThrowIfNull(query);

          // 1) Extract claimed_id and its 17-digit SteamID.
          var claimedId = query["openid.claimed_id"].ToString();
          if (string.IsNullOrEmpty(claimedId))
              return SteamVerificationResult.Invalid("claimed_id_missing");

          var match = SteamConstants.ClaimedIdRegex().Match(claimedId);
          if (!match.Success)
              return SteamVerificationResult.Invalid("claimed_id_malformed");

          // 2) Build check_authentication payload — echo EVERY openid.* param, set mode.
          var form = new List&lt;KeyValuePair&lt;string, string&gt;&gt;();
          foreach (var kv in query)
          {
              if (kv.Key.StartsWith("openid.", StringComparison.Ordinal))
                  form.Add(new KeyValuePair&lt;string, string&gt;(kv.Key, kv.Value.ToString()));
          }
          // Replace or add openid.mode = check_authentication.
          form.RemoveAll(kv =&gt; kv.Key == "openid.mode");
          form.Add(new KeyValuePair&lt;string, string&gt;("openid.mode", "check_authentication"));

          using var req = new HttpRequestMessage(HttpMethod.Post, _opts.Steam.OpenIdEndpoint)
          {
              Content = new FormUrlEncodedContent(form),
          };
          using var resp = await _httpClient.SendAsync(req, cancellationToken).ConfigureAwait(false);
          if (!resp.IsSuccessStatusCode)
              return SteamVerificationResult.Invalid("check_authentication_http_error");

          // 3) Parse Key-Value form.
          var body = await resp.Content.ReadAsStringAsync(cancellationToken).ConfigureAwait(false);
          var isValid = IsValidTrueInBody(body);

          return isValid
              ? SteamVerificationResult.Ok(match.Groups[1].Value)
              : SteamVerificationResult.Invalid("is_valid_false");
      }

      private static bool IsValidTrueInBody(string body)
      {
          foreach (var line in body.Split('\n'))
          {
              if (line.TrimEnd('\r') == SteamConstants.IsValidTrueLine)
                  return true;
          }
          return false;
      }
  }
  ```

  ```csharp
  // SPDX-License-Identifier: GPL-3.0-or-later
  // Copyright (c) 2026 GameKit contributors

  using System;
  using System.Threading;
  using System.Threading.Tasks;
  using GameKit.Auth.Entities;
  using GameKit.Auth.Services;
  using GameKit.Core.Data;
  using GameKit.Core.Entities;
  using GameKit.Core.Services;
  using Microsoft.EntityFrameworkCore;

  namespace GameKit.Auth.Providers.Steam;

  /// &lt;summary&gt;
  /// Steam OAuth provider (AUTH-06). Upserts the &lt;c&gt;Player&lt;/c&gt; + &lt;c&gt;PlayerIdentity&lt;/c&gt; pair for the
  /// verified Steam ID (verification happens in the endpoint via &lt;see cref="SteamOpenIdVerifier"/&gt;)
  /// and issues a root refresh-token family. This type does NOT trust the caller's &lt;c&gt;externalId&lt;/c&gt;
  /// as a claim of authenticity — it assumes the caller has already run &lt;c&gt;check_authentication&lt;/c&gt;.
  /// &lt;/summary&gt;
  internal sealed class SteamOAuthProvider : IOAuthProvider
  {
      private readonly GameKitDbContext _ctx;
      private readonly IClock _clock;
      private readonly IIdGenerator _ids;
      private readonly IRefreshTokenService _refresh;

      /// &lt;summary&gt;Constructs the provider.&lt;/summary&gt;
      public SteamOAuthProvider(GameKitDbContext ctx, IClock clock, IIdGenerator ids, IRefreshTokenService refresh)
      {
          _ctx = ctx; _clock = clock; _ids = ids; _refresh = refresh;
      }

      /// &lt;inheritdoc /&gt;
      public string Provider =&gt; "steam";

      /// &lt;inheritdoc /&gt;
      public async Task&lt;OAuthResult&gt; CompleteLoginAsync(
          string externalId,
          string? displayName,
          string? avatarUrl,
          string? fingerprint,
          CancellationToken cancellationToken = default)
      {
          // Upsert: look up PlayerIdentity by (steam, externalId). If found, reuse its PlayerId; else create both.
          var existing = await _ctx.Set&lt;PlayerIdentity&gt;()
              .AsNoTracking()
              .FirstOrDefaultAsync(i =&gt; i.Provider == Provider &amp;&amp; i.ExternalId == externalId, cancellationToken)
              .ConfigureAwait(false);

          Guid playerId;
          if (existing is not null)
          {
              playerId = existing.PlayerId;
              // Optionally refresh display name / avatar on subsequent login.
              var tracked = await _ctx.Set&lt;PlayerIdentity&gt;().FirstAsync(i =&gt; i.Id == existing.Id, cancellationToken).ConfigureAwait(false);
              tracked.DisplayName = displayName ?? tracked.DisplayName;
              tracked.AvatarUrl = avatarUrl ?? tracked.AvatarUrl;
              tracked.UpdatedAt = _clock.UtcNow;
              await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
          }
          else
          {
              playerId = _ids.NewId();
              _ctx.Players.Add(new Player
              {
                  Id = playerId,
                  DisplayName = displayName ?? $"SteamUser-{externalId[^8..]}",
                  CreatedAt = _clock.UtcNow,
              });
              _ctx.Set&lt;PlayerIdentity&gt;().Add(new PlayerIdentity
              {
                  Id = _ids.NewId(),
                  PlayerId = playerId,
                  Provider = Provider,
                  ExternalId = externalId,
                  DisplayName = displayName,
                  AvatarUrl = avatarUrl,
                  CreatedAt = _clock.UtcNow,
                  UpdatedAt = _clock.UtcNow,
              });
              await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
          }

          var tokens = await _refresh.IssueRootAsync(playerId, Provider, fingerprint, cancellationToken).ConfigureAwait(false);
          return OAuthResult.Ok(playerId, tokens);
      }
  }
  ```

  ```csharp
  /// &lt;summary&gt;
  /// Discord OAuth2 provider (AUTH-07). Consumes the Discord snowflake + username from the aspnet-contrib
  /// handler's &lt;c&gt;OnCreatingTicket&lt;/c&gt; event; performs the Player + PlayerIdentity upsert and issues the
  /// refresh-token family. Scope is locked to &lt;c&gt;identify&lt;/c&gt; at the handler layer (Task 2 (d)).
  /// &lt;/summary&gt;
  internal sealed class DiscordOAuthProvider : IOAuthProvider
  {
      private readonly GameKitDbContext _ctx;
      private readonly IClock _clock;
      private readonly IIdGenerator _ids;
      private readonly IRefreshTokenService _refresh;

      /// &lt;summary&gt;Constructs the provider.&lt;/summary&gt;
      public DiscordOAuthProvider(GameKitDbContext ctx, IClock clock, IIdGenerator ids, IRefreshTokenService refresh)
      {
          _ctx = ctx; _clock = clock; _ids = ids; _refresh = refresh;
      }

      /// &lt;inheritdoc /&gt;
      public string Provider =&gt; "discord";

      /// &lt;inheritdoc /&gt;
      public async Task&lt;OAuthResult&gt; CompleteLoginAsync(
          string externalId,
          string? displayName,
          string? avatarUrl,
          string? fingerprint,
          CancellationToken cancellationToken = default)
      {
          var existing = await _ctx.Set&lt;PlayerIdentity&gt;()
              .AsNoTracking()
              .FirstOrDefaultAsync(i =&gt; i.Provider == Provider &amp;&amp; i.ExternalId == externalId, cancellationToken)
              .ConfigureAwait(false);

          Guid playerId;
          if (existing is not null)
          {
              playerId = existing.PlayerId;
              var tracked = await _ctx.Set&lt;PlayerIdentity&gt;().FirstAsync(i =&gt; i.Id == existing.Id, cancellationToken).ConfigureAwait(false);
              tracked.DisplayName = displayName ?? tracked.DisplayName;
              tracked.AvatarUrl = avatarUrl ?? tracked.AvatarUrl;
              tracked.UpdatedAt = _clock.UtcNow;
              await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
          }
          else
          {
              playerId = _ids.NewId();
              _ctx.Players.Add(new Player
              {
                  Id = playerId,
                  DisplayName = displayName ?? $"DiscordUser-{externalId[^6..]}",
                  CreatedAt = _clock.UtcNow,
              });
              _ctx.Set&lt;PlayerIdentity&gt;().Add(new PlayerIdentity
              {
                  Id = _ids.NewId(),
                  PlayerId = playerId,
                  Provider = Provider,
                  ExternalId = externalId,
                  DisplayName = displayName,
                  AvatarUrl = avatarUrl,
                  CreatedAt = _clock.UtcNow,
                  UpdatedAt = _clock.UtcNow,
              });
              await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
          }

          var tokens = await _refresh.IssueRootAsync(playerId, Provider, fingerprint, cancellationToken).ConfigureAwait(false);
          return OAuthResult.Ok(playerId, tokens);
      }
  }
  ```

| Boundary | Description |
|----------|-------------|
| browser → /auth/callback/steam | Browser-provided `openid.*` query params are untrusted until `check_authentication` succeeds |
