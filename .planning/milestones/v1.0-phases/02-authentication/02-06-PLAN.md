  ```csharp
  // SPDX-License-Identifier: GPL-3.0-or-later
  // Copyright (c) 2026 GameKit contributors

  using System;
  using System.Threading;
  using System.Threading.Tasks;
  using GameKit.Auth.Entities;
  using GameKit.Auth.Providers;
  using GameKit.Auth.Services;
  using GameKit.Core.Data;
  using GameKit.Core.Entities;
  using GameKit.Core.Services;
  using Microsoft.EntityFrameworkCore;

  namespace GameKit.Auth.Providers.Password;

  /// &lt;summary&gt;
  /// Username+password provider (AUTH-09). Hashes via &lt;see cref="IPasswordHasher"/&gt; (BCrypt default;
  /// Argon2 swappable per AUTH-16). Runs a dummy verify on user-not-found to equalize timing
  /// (mitigation of T-02-16 flagged in plan 02-04 follow-up).
  /// &lt;/summary&gt;
  internal sealed class PasswordOAuthProvider : IOAuthProvider
  {
      // Stable hash over "always-fails-never-matches" used to equalize timing when username lookup misses.
      private const string DummyHash = "$2a$12$abcdefghijklmnopqrstuu1234567890123456789012345678ab";

      private readonly GameKitDbContext _ctx;
      private readonly IClock _clock;
      private readonly IIdGenerator _ids;
      private readonly IPasswordHasher _hasher;
      private readonly IRefreshTokenService _refresh;
      private readonly IAuthAuditWriter _audit;

      public PasswordOAuthProvider(
          GameKitDbContext ctx, IClock clock, IIdGenerator ids,
          IPasswordHasher hasher, IRefreshTokenService refresh, IAuthAuditWriter audit)
      {
          _ctx = ctx; _clock = clock; _ids = ids;
          _hasher = hasher; _refresh = refresh; _audit = audit;
      }

      /// &lt;inheritdoc /&gt;
      public string Provider =&gt; "password";

      /// &lt;summary&gt;Login: &lt;paramref name="externalId"/&gt; = username; &lt;paramref name="displayName"/&gt; = password (by convention; the endpoint layer enforces).&lt;/summary&gt;
      /// &lt;inheritdoc /&gt;
      public async Task&lt;OAuthResult&gt; CompleteLoginAsync(
          string externalId,
          string? displayName,
          string? avatarUrl,
          string? fingerprint,
          CancellationToken cancellationToken = default)
      {
          // "externalId" is the username; "displayName" is the password — contract from /auth/login/password endpoint.
          var username = externalId;
          var password = displayName ?? string.Empty;

          var credential = await _ctx.Set&lt;PlayerCredential&gt;()
              .AsNoTracking()
              .FirstOrDefaultAsync(c =&gt; c.Username == username, cancellationToken)
              .ConfigureAwait(false);

          if (credential is null)
          {
              // Timing-attack mitigation: run BCrypt.Verify against a known-bad hash so wall-clock parity with hit case.
              _ = _hasher.Verify(password, DummyHash);
              await _audit.WriteAsync("auth.login.failure", "player", null, null, new { provider = "password", reason_code = "unknown_username" }, null, cancellationToken).ConfigureAwait(false);
              return OAuthResult.Fail("invalid_credentials");
          }

          if (!_hasher.Verify(password, credential.PasswordHash))
          {
              await _audit.WriteAsync("auth.login.failure", "player", credential.PlayerId, credential.PlayerId, new { provider = "password", reason_code = "wrong_password" }, null, cancellationToken).ConfigureAwait(false);
              return OAuthResult.Fail("invalid_credentials");
          }

          var tokens = await _refresh.IssueRootAsync(credential.PlayerId, Provider, fingerprint, cancellationToken).ConfigureAwait(false);
          return OAuthResult.Ok(credential.PlayerId, tokens);
      }

      /// &lt;summary&gt;
      /// Registers a new player with a username+password credential. Called from &lt;c&gt;/auth/register&lt;/c&gt;
      /// when no guest JWT is present (D-12 upgrade-in-place path is handled by &lt;see cref="IGuestUpgradeService"/&gt;).
      /// &lt;/summary&gt;
      public async Task&lt;OAuthResult&gt; RegisterAsync(
          string username, string password, string? displayName, string? fingerprint,
          CancellationToken cancellationToken = default)
      {
          // Concurrent same-username collisions surface here as Postgres 23505 (unique index on player_credentials.username)
          // — RESEARCH §15 open question #3. Caller must translate to HTTP 409 `username_taken`.
          var playerId = _ids.NewId();
          var display = string.IsNullOrWhiteSpace(displayName) ? username : displayName!;
          _ctx.Players.Add(new Player
          {
              Id = playerId,
              DisplayName = display,
              CreatedAt = _clock.UtcNow,
          });
          _ctx.Set&lt;PlayerCredential&gt;().Add(new PlayerCredential
          {
              PlayerId = playerId,
              Username = username,
              PasswordHash = _hasher.Hash(password),
              UpdatedAt = _clock.UtcNow,
          });

          try
          {
              await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
          }
          catch (DbUpdateException ex) when (ex.InnerException is Npgsql.PostgresException pg &amp;&amp; pg.SqlState == "23505")
          {
              await _audit.WriteAsync("auth.login.failure", "player", null, null, new { provider = "password", reason_code = "username_taken" }, null, cancellationToken).ConfigureAwait(false);
              return OAuthResult.Fail("username_taken");
          }

          await _audit.WriteAsync("auth.credential.password_set", "player_credential", playerId, playerId, new { password_set = true, work_factor = 12 }, null, cancellationToken).ConfigureAwait(false);

          var tokens = await _refresh.IssueRootAsync(playerId, Provider, fingerprint, cancellationToken).ConfigureAwait(false);
          return OAuthResult.Ok(playerId, tokens);
      }
  }
  ```

  ```csharp
  namespace GameKit.Auth.Services;

  /// &lt;summary&gt;Outcome of &lt;see cref="IIdentityLinker.LinkAsync"/&gt;.&lt;/summary&gt;
  public enum LinkResultKind
  {
      /// &lt;summary&gt;Identity was newly linked to the caller's player.&lt;/summary&gt;
      Linked,
      /// &lt;summary&gt;Identity was already linked to the caller's own player — idempotent no-op.&lt;/summary&gt;
      AlreadyLinkedToSelf,
      /// &lt;summary&gt;Identity is linked to a DIFFERENT player — return 409 with &lt;see cref="ExternalIdHash"/&gt; in body (D-11).&lt;/summary&gt;

  /// The three possible outcomes of an &lt;see cref="IIdentityLinker.LinkAsync"/&gt; call.
  /// When &lt;see cref="Kind"/&gt; is &lt;see cref="LinkResultKind.AlreadyLinkedToOtherPlayer"/&gt;,
  /// &lt;see cref="ExternalIdHash"/&gt; carries the SHA-256 hash that the 409 response body exposes

  /// &lt;/summary&gt;
  public sealed record LinkResult(LinkResultKind Kind, string? ExternalIdHash)
  {
      /// &lt;summary&gt;Convenience factory.&lt;/summary&gt;
      public static LinkResult Linked() =&gt; new(LinkResultKind.Linked, null);
      /// &lt;summary&gt;Convenience factory.&lt;/summary&gt;
      public static LinkResult AlreadyLinkedToSelf() =&gt; new(LinkResultKind.AlreadyLinkedToSelf, null);
      /// &lt;summary&gt;Convenience factory.&lt;/summary&gt;
      public static LinkResult AlreadyLinkedToOtherPlayer(string externalIdHash) =&gt; new(LinkResultKind.AlreadyLinkedToOtherPlayer, externalIdHash);
  ```

  ```csharp
  // SPDX-License-Identifier: GPL-3.0-or-later
  // Copyright (c) 2026 GameKit contributors

  using System;
  using System.Data;
  using System.Threading;
  using System.Threading.Tasks;
  using GameKit.Auth.Entities;
  using GameKit.Core.Data;
  using GameKit.Core.Services;
  using Microsoft.EntityFrameworkCore;
  using Npgsql;

  namespace GameKit.Auth.Services;

  /// &lt;summary&gt;Default &lt;see cref="IIdentityLinker"/&gt;. SERIALIZABLE tx + 40001 retry + 23505 hard-fail.&lt;/summary&gt;
  internal sealed class IdentityLinker : IIdentityLinker
  {
      private const int MaxRetries = 3;

      private readonly GameKitDbContext _ctx;
      private readonly IClock _clock;
      private readonly IIdGenerator _ids;
      private readonly IExternalIdHasher _hasher;
      private readonly IAuthAuditWriter _audit;

      public IdentityLinker(
          GameKitDbContext ctx, IClock clock, IIdGenerator ids,
          IExternalIdHasher hasher, IAuthAuditWriter audit)
      {
          _ctx = ctx; _clock = clock; _ids = ids;
          _hasher = hasher; _audit = audit;
      }

      /// &lt;inheritdoc /&gt;
      public async Task&lt;LinkResult&gt; LinkAsync(Guid playerId, string provider, string externalId, CancellationToken cancellationToken = default)
      {
          for (int attempt = 0; attempt &lt; MaxRetries; attempt++)
          {
              await using var tx = await _ctx.Database
                  .BeginTransactionAsync(IsolationLevel.Serializable, cancellationToken)
                  .ConfigureAwait(false);
              try
              {
                  var existing = await _ctx.Set&lt;PlayerIdentity&gt;()
                      .AsNoTracking()
                      .FirstOrDefaultAsync(pi =&gt; pi.Provider == provider &amp;&amp; pi.ExternalId == externalId, cancellationToken)
                      .ConfigureAwait(false);

                  if (existing is not null &amp;&amp; existing.PlayerId != playerId)
                  {
                      var hash = _hasher.Hash(provider, externalId);
                      await _audit.WriteAsync(
                          "auth.identity.link_failed_collision", "player", playerId, playerId,
                          new { provider, external_id_hash = hash },
                          "cross_player_collision", cancellationToken).ConfigureAwait(false);
                      await tx.CommitAsync(cancellationToken).ConfigureAwait(false);   // commit audit row
                      return LinkResult.AlreadyLinkedToOtherPlayer(hash);
                  }

                  if (existing is not null)
                  {
                      // Idempotent: already linked to me.
                      await tx.CommitAsync(cancellationToken).ConfigureAwait(false);
                      return LinkResult.AlreadyLinkedToSelf();
                  }

                  _ctx.Set&lt;PlayerIdentity&gt;().Add(new PlayerIdentity
                  {
                      Id = _ids.NewId(),
                      PlayerId = playerId,
                      Provider = provider,
                      ExternalId = externalId,
                      CreatedAt = _clock.UtcNow,
                      UpdatedAt = _clock.UtcNow,
                  });
                  await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);

                  await _audit.WriteAsync(
                      "auth.identity.linked", "player_identity", playerId, playerId,
                      new { provider, external_id_hash = _hasher.Hash(provider, externalId) },
                      null, cancellationToken).ConfigureAwait(false);

                  await tx.CommitAsync(cancellationToken).ConfigureAwait(false);
                  return LinkResult.Linked();
              }
              catch (DbUpdateException ex) when (ex.InnerException is PostgresException pg)
              {
                  await tx.RollbackAsync(cancellationToken).ConfigureAwait(false);

                  if (pg.SqlState == "23505")
                  {
                      // Lost the D-14 race. Caller's client is told "already linked" with a hash.
                      var hash = _hasher.Hash(provider, externalId);
                      await _audit.WriteAsync(
                          "auth.identity.link_failed_collision", "player", playerId, playerId,
                          new { provider, external_id_hash = hash },
                          "cross_player_collision", cancellationToken).ConfigureAwait(false);
                      return LinkResult.AlreadyLinkedToOtherPlayer(hash);
                  }

                  if (pg.SqlState == "40001" &amp;&amp; attempt &lt; MaxRetries - 1)
                      continue;   // serialization_failure — retry.

                  throw;
              }
          }
          throw new InvalidOperationException("IdentityLinker: SERIALIZABLE retries exhausted.");
  ```

  ```csharp
  /// &lt;summary&gt;Default &lt;see cref="IGuestUpgradeService"/&gt;.&lt;/summary&gt;
  internal sealed class GuestUpgradeService : IGuestUpgradeService
  {
      private const int MaxRetries = 3;

      private readonly GameKitDbContext _ctx;
      private readonly IClock _clock;
      private readonly IPasswordHasher _hasher;
      private readonly IRefreshTokenService _refresh;
      private readonly IAuthAuditWriter _audit;
      private readonly IIdentityLinker _identityLinker;

      public GuestUpgradeService(
          GameKitDbContext ctx, IClock clock, IPasswordHasher hasher,
          IRefreshTokenService refresh, IAuthAuditWriter audit, IIdentityLinker identityLinker)
      {
          _ctx = ctx; _clock = clock; _hasher = hasher;
          _refresh = refresh; _audit = audit; _identityLinker = identityLinker;
      }

      /// &lt;inheritdoc /&gt;
      public async Task&lt;TokenPair&gt; UpgradeToPasswordAsync(Guid playerId, string username, string password, string? fingerprint, CancellationToken cancellationToken = default)
      {
          for (int attempt = 0; attempt &lt; MaxRetries; attempt++)
          {
              await using var tx = await _ctx.Database
                  .BeginTransactionAsync(IsolationLevel.Serializable, cancellationToken)
                  .ConfigureAwait(false);
              try
              {
                  _ctx.Set&lt;PlayerCredential&gt;().Add(new PlayerCredential
                  {
                      PlayerId = playerId,
                      Username = username,
                      PasswordHash = _hasher.Hash(password),
                      UpdatedAt = _clock.UtcNow,
                  });
                  await _ctx.SaveChangesAsync(cancellationToken).ConfigureAwait(false);

                  await _audit.WriteAsync(
                      "auth.guest.upgraded_password", "player", playerId, playerId,
                      new { is_guest = false, credential_added = true },
                      null, cancellationToken).ConfigureAwait(false);

                  await tx.CommitAsync(cancellationToken).ConfigureAwait(false);

                  // Issue a fresh root token (no longer a guest — IsGuestResolver now returns false).
                  return await _refresh.IssueRootAsync(playerId, "password", fingerprint, cancellationToken).ConfigureAwait(false);
              }
              catch (DbUpdateException ex) when (ex.InnerException is PostgresException pg)
              {
                  await tx.RollbackAsync(cancellationToken).ConfigureAwait(false);

                  if (pg.SqlState == "23505")
                      throw new UsernameAlreadyTakenException(username);

                  if (pg.SqlState == "40001" &amp;&amp; attempt &lt; MaxRetries - 1)
                      continue;

                  throw;
              }
          }
          throw new InvalidOperationException("GuestUpgradeService: SERIALIZABLE retries exhausted.");
      }

      /// &lt;inheritdoc /&gt;
      public Task&lt;LinkResult&gt; UpgradeToLinkedOAuthAsync(Guid playerId, string provider, string externalId, CancellationToken cancellationToken = default) =&gt;
          _identityLinker.LinkAsync(playerId, provider, externalId, cancellationToken);
  ```
