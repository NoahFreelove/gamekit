    ```csharp
    // SPDX-License-Identifier: GPL-3.0-or-later
    // Copyright (c) 2026 GameKit contributors
    using System;
    using System.Security.Cryptography;
    using System.Threading.Tasks;
    using Microsoft.AspNetCore.Http;

    namespace GameKit.Admin.UI.Middleware;

    /// <summary>
    /// Per-request CSP nonce middleware (D-15 / ADMIN-12). For every request whose path starts with
    /// <see cref="GameKitAdminOptions.MountPath"/>, generates a 128-bit cryptographically random
    /// nonce, stashes it in <c>HttpContext.Items["gamekit.admin.csp-nonce"]</c>, and emits a strict
    /// <c>Content-Security-Policy</c> header on the response. Non-admin paths pass through unchanged.
    /// </summary>
    public sealed class AdminCspNonceMiddleware
    {
        /// <summary>Key under which the per-request nonce is stored in <see cref="HttpContext.Items"/>.</summary>
        public const string NonceItemKey = "gamekit.admin.csp-nonce";

        private readonly RequestDelegate _next;
        private readonly PathString _adminPrefix;

        /// <summary>Resolved from DI: the next middleware + the current admin options.</summary>
        public AdminCspNonceMiddleware(RequestDelegate next, GameKitAdminOptions options)
        {
            ArgumentNullException.ThrowIfNull(next);
            ArgumentNullException.ThrowIfNull(options);
            _next = next;
            _adminPrefix = options.MountPath;
        }

        /// <inheritdoc />
        public async Task InvokeAsync(HttpContext ctx)
        {
            if (!ctx.Request.Path.StartsWithSegments(_adminPrefix))
            {
                await _next(ctx).ConfigureAwait(false);
                return;
            }

            // Generate 128-bit nonce, base64 encode.
            Span<byte> nonceBytes = stackalloc byte[16];
            RandomNumberGenerator.Fill(nonceBytes);
            var nonce = Convert.ToBase64String(nonceBytes);
            ctx.Items[NonceItemKey] = nonce;

            // Emit CSP header just before the response starts (OnStarting runs at header flush).
            ctx.Response.OnStarting(() =>
            {
                if (!ctx.Response.Headers.ContainsKey("Content-Security-Policy"))
                {
                    ctx.Response.Headers["Content-Security-Policy"] =
                        "default-src 'self'; " +
                        $"script-src 'self' 'nonce-{nonce}'; " +
                        "style-src 'self' 'unsafe-inline'; " +
                        "img-src 'self' data:; " +
                        "font-src 'self'; " +
                        "connect-src 'self'; " +
                        "frame-ancestors 'none'; " +
                        "base-uri 'self'; " +
                        "form-action 'self'";
                }
                return Task.CompletedTask;
            });

            await _next(ctx).ConfigureAwait(false);
        }
    }
    ```

    Step 2 — Implement `src/GameKit.Admin.UI/Http/EndpointFilters/AntiforgeryValidationFilter.cs`:

    ```csharp
    // SPDX-License-Identifier: GPL-3.0-or-later
    // Copyright (c) 2026 GameKit contributors
    using System;
    using System.Threading.Tasks;
    using Microsoft.AspNetCore.Antiforgery;
    using Microsoft.AspNetCore.Http;
    using Microsoft.Extensions.DependencyInjection;

    namespace GameKit.Admin.UI.Http.EndpointFilters;

    /// <summary>
    /// Endpoint filter that validates the antiforgery token on every request (D-16 / ADMIN-12).
    /// Resolves <see cref="IAntiforgery"/> from the request services (not a constructor parameter —
    /// the filter is stateless). Failure returns <c>400 BadRequest</c> with <c>error=csrf_validation_failed</c>;
    /// plain-minimal-api-consumable JSON.
    /// </summary>
    public sealed class AntiforgeryValidationFilter : IEndpointFilter
    {
        /// <inheritdoc />
        public async ValueTask<object?> InvokeAsync(
            EndpointFilterInvocationContext context,
            EndpointFilterDelegate next)
        {
            ArgumentNullException.ThrowIfNull(context);
            ArgumentNullException.ThrowIfNull(next);

            var antiforgery = context.HttpContext.RequestServices.GetRequiredService<IAntiforgery>();
            try
            {
                await antiforgery.ValidateRequestAsync(context.HttpContext).ConfigureAwait(false);
            }
            catch (AntiforgeryValidationException)
            {
                return Results.BadRequest(new { error = "csrf_validation_failed" });
            }
            return await next(context).ConfigureAwait(false);
        }
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| admin browser → CSP enforcement | Browser enforces CSP client-side; assumption is a modern browser. Attackers running a scriptable agent bypass CSP but other layers (auth cookie + antiforgery) still apply. |
| admin browser → antiforgery token | JS reads `gk_admin_csrf` cookie + echoes to `X-GameKit-Admin-CSRF` header. Attacker-controlled origin cannot forge due to SameSite=Lax + HTTPS. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-05-01 | Tampering | XSS via MudBlazor inline script | mitigate | CSP `script-src 'self' 'nonce-<per-request>'` — any inline `<script>` without the nonce is blocked. MudBlazor 9.3.0 confirmed ships NO inline scripts (RESEARCH A6). |
| T-03-05-02 | Tampering | Clickjacking via iframe embedding | mitigate | CSP `frame-ancestors 'none'` — browsers block framing unconditionally. |
| T-03-05-03 | Spoofing | CSRF on mutation endpoints | mitigate | `AntiforgeryValidationFilter` + `IAntiforgery.ValidateRequestAsync` on every POST/DELETE/PATCH admin endpoint; request without token → 400. |
| T-03-05-04 | Information Disclosure | CSP nonce leaks to non-admin pages if middleware mis-scoped | mitigate | Middleware exits early for non-admin paths (unit test `NonAdminPath_NoNonce_NoCspHeader`). |
| T-03-05-05 | Tampering | attacker sniffs response header and replays nonce | accept | Nonce has single-request scope; replaying a nonce in a script tag served by the attacker's origin is pointless (browser won't execute from a different origin). |
| T-03-05-06 | Information Disclosure | CSP report-uri leaks admin panel topology | accept/defer | No `report-uri` or `report-to` directive emitted in v1 (library does not phone home, CONTEXT D-15 + deferred list). Optional `report-only` header toggle exists via `GameKitAdminOptions.Csp.ReportOnly` but consumer must own the endpoint. |
| T-03-05-07 | Tampering | predictable nonce allows bypass | mitigate | `RandomNumberGenerator.Fill(stackalloc byte[16])` — 128 bits of cryptographic entropy per request. |
</threat_model>
