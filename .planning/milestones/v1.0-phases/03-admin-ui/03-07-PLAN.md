    public sealed class BanPlayerRequestValidator : AbstractValidator<BanPlayerRequest>
    {
        public BanPlayerRequestValidator()
        {
            RuleFor(x => x.Reason)
                .NotEmpty().WithMessage("A reason is required.")
                .MinimumLength(3).WithMessage("Reason must be at least 3 characters.")
                .MaximumLength(512).WithMessage("Reason is too long (max 512 characters).");
        }
    }

    // CreateAdminRequestValidator.cs — username regex + role enum + password min length
    public sealed class CreateAdminRequestValidator : AbstractValidator<CreateAdminRequest>
    {
        private static readonly Regex UsernameRegex = new("^[a-z0-9_-]{3,32}$", RegexOptions.Compiled);
        public CreateAdminRequestValidator()
        {
            RuleFor(x => x.Username).NotEmpty()
                .Matches(UsernameRegex).WithMessage("Username must be 3–32 chars, lowercase + digits + underscore/hyphen.");
            RuleFor(x => x.Password).NotEmpty()
                .MinimumLength(8).WithMessage("Password must be at least 8 characters.");
            RuleFor(x => x.Role).Must(r => r == AdminRoles.Admin || r == AdminRoles.Superadmin)
                .WithMessage("Role must be 'admin' or 'superadmin'.");
        }
    }

    // PlayerSearchRequestValidator.cs — presence + page size bounds
    public sealed class PlayerSearchRequestValidator : AbstractValidator<PlayerSearchRequest>
    {
        public PlayerSearchRequestValidator()
        {
            RuleFor(x => x.Query).NotEmpty().MaximumLength(256);
            RuleFor(x => x.PageSize).InclusiveBetween(1, 50);
        }
    }

    // LoginRequestValidator.cs — presence only; no content rules (BCrypt handles the rest)

    // SPDX-License-Identifier: GPL-3.0-or-later
    // Copyright (c) 2026 GameKit contributors
    namespace GameKit.Admin.UI.Http;

    /// <summary>
    /// Maps the <c>/admin/api/*</c> minimal-API surface onto a <see cref="RouteGroupBuilder"/>.
    /// Called by <c>MapGameKitAdmin(prefix)</c> — see <see cref="AdminApplicationBuilderExtensions"/>.
    /// </summary>
    internal static class AdminEndpoints
    {
        public static RouteGroupBuilder Map(RouteGroupBuilder group)
        {
            // POST /login — anonymous, rate-limited, validator
            group.MapPost("/login", LoginAsync)
                .RequireRateLimiting(AdminRateLimitRegistrations.AdminLoginPolicy)
                .AddEndpointFilter<ValidationEndpointFilter<LoginRequest>>()
                .AllowAnonymous();

            // POST /logout — clear cookie; anonymous (cookie presence is the auth factor)
            group.MapPost("/logout", LogoutAsync).AllowAnonymous();

            // GET /players/search — Admin; read-only, so NO antiforgery filter (W8: D-16 antiforgery applies to
            // mutations only — search is idempotent and carries no side effects). Query-string parameters bind
            // via [AsParameters] into PlayerSearchRequest.
            group.MapGet("/players/search", SearchPlayersAsync)
                .RequireAuthorization(AdminPolicies.Admin)
                .AddEndpointFilter<ValidationEndpointFilter<PlayerSearchRequest>>();

            // POST /players/{id}/ban — Admin, antiforgery + validator
            group.MapPost("/players/{id:guid}/ban", BanPlayerAsync)
                .RequireAuthorization(AdminPolicies.Admin)
                .AddEndpointFilter<AntiforgeryValidationFilter>()
                .AddEndpointFilter<ValidationEndpointFilter<BanPlayerRequest>>();

            // POST /players/{id}/unban — Admin, antiforgery (no validator — reason optional)
            group.MapPost("/players/{id:guid}/unban", UnbanPlayerAsync)
                .RequireAuthorization(AdminPolicies.Admin)
                .AddEndpointFilter<AntiforgeryValidationFilter>();

            // POST /players/{id}/gdpr-delete — Superadmin, antiforgery
            group.MapPost("/players/{id:guid}/gdpr-delete", GdprDeletePlayerAsync)
                .RequireAuthorization(AdminPolicies.Superadmin)
                .AddEndpointFilter<AntiforgeryValidationFilter>();

            // GET /admins — Superadmin list
            group.MapGet("/admins", ListAdminsAsync)
                .RequireAuthorization(AdminPolicies.Superadmin);

            // POST /admins — Superadmin create
            group.MapPost("/admins", CreateAdminAsync)
                .RequireAuthorization(AdminPolicies.Superadmin)
                .AddEndpointFilter<AntiforgeryValidationFilter>()
                .AddEndpointFilter<ValidationEndpointFilter<CreateAdminRequest>>();

            // DELETE /admins/{id} — Superadmin
            group.MapDelete("/admins/{id:guid}", DeleteAdminAsync)
                .RequireAuthorization(AdminPolicies.Superadmin)
                .AddEndpointFilter<AntiforgeryValidationFilter>();

            // GET /audit — Admin paginated
            group.MapGet("/audit", GetAuditAsync).RequireAuthorization(AdminPolicies.Admin);

            // GET /match-history — Admin paginated
            group.MapGet("/match-history", GetMatchHistoryAsync).RequireAuthorization(AdminPolicies.Admin);

            // GET /health — Admin (3-probe report)
            group.MapGet("/health", GetHealthAsync).RequireAuthorization(AdminPolicies.Admin);

            return group;
        }

        // Handlers:
        private static async Task<IResult> LoginAsync(
            LoginRequest req, HttpContext http, IAdminAuthService authSvc, CancellationToken ct)
        {
            var result = await authSvc.VerifyPasswordAsync(req.Username, req.Password, ct);
            if (result is null) return Results.Unauthorized();
            var (adminId, role) = result.Value;

            var claims = new List<Claim>
            {
                new(ClaimTypes.NameIdentifier, adminId.ToString()),
                new(ClaimTypes.Name, req.Username),
                new(ClaimTypes.Role, role),
            };
            var identity = new ClaimsIdentity(claims, AdminAuthenticationSchemeConstants.Scheme);
            var principal = new ClaimsPrincipal(identity);

            var authProps = new AuthenticationProperties
            {
                IsPersistent = req.RememberMe,
                ExpiresUtc = req.RememberMe ? DateTimeOffset.UtcNow.AddDays(30) : null,
            };
            await http.SignInAsync(AdminAuthenticationSchemeConstants.Scheme, principal, authProps);
            return Results.Ok(new { success = true, redirectUrl = "/admin" });
        }

        private static async Task<IResult> LogoutAsync(HttpContext http)
        {
            await http.SignOutAsync(AdminAuthenticationSchemeConstants.Scheme);
            return Results.Ok(new { success = true, redirectUrl = "/admin/login" });
        }

        private static async Task<IResult> SearchPlayersAsync(
            [AsParameters] PlayerSearchRequest req, IPlayerSearchService svc, CancellationToken ct)
        {
            var result = await svc.SearchAsync(req.Query, req.AfterId, req.PageSize, ct);
            return Results.Ok(result);
        }

        private static async Task<IResult> BanPlayerAsync(
            Guid id, BanPlayerRequest req, HttpContext http, IPlayerBanService bans, CancellationToken ct)
        {
            var actorId = GetAdminId(http);
            await bans.BanAsync(id, actorId, req.Reason, ct);
            return Results.NoContent();
        }

        private static async Task<IResult> UnbanPlayerAsync(
            Guid id, UnbanPlayerRequest req, HttpContext http, IPlayerBanService bans, CancellationToken ct)
        {
            var actorId = GetAdminId(http);
            await bans.UnbanAsync(id, actorId, req.Reason, ct);
            return Results.NoContent();
        }

        private static async Task<IResult> GdprDeletePlayerAsync(
            Guid id, GdprDeleteRequest req, HttpContext http,
            IGdprDeleteService gdpr, IAdminAuditWriter audit, CancellationToken ct)
        {
            var actorId = GetAdminId(http);
            await gdpr.DeletePlayerAsync(id, actorId, req.Reason ?? "gdpr_request", ct);
            await audit.WriteAsync(AdminAuditActions.PlayerGdprDelete, "player", id, actorId,
                null, new { deleted = true }, req.Reason, ct);
            return Results.NoContent();
        }

        private static async Task<IResult> ListAdminsAsync(IAdminUserService svc, CancellationToken ct)
            => Results.Ok(await svc.ListAsync(afterId: null, pageSize: 50, ct));

        private static async Task<IResult> CreateAdminAsync(
            CreateAdminRequest req, HttpContext http, IAdminUserService svc, CancellationToken ct)
        {
            var actorId = GetAdminId(http);
            var id = await svc.CreateAsync(req.Username, req.Password, req.Role, actorId, ct);
            return Results.Ok(new { id });
        }

        private static async Task<IResult> DeleteAdminAsync(
            Guid id, HttpContext http, IAdminUserService svc, CancellationToken ct)
        {
            var actorId = GetAdminId(http);
            await svc.DeleteAsync(id, actorId, ct);
            return Results.NoContent();
        }

        private static async Task<IResult> GetAuditAsync(
            [AsParameters] AuditQuery q, GameKitDbContext ctx, CancellationToken ct)
        {
            // Keyset pagination — WHERE (created_at, id) < (afterCreatedAt, afterId); take pageSize+1
            var page = q.PageSize ?? 50;
            var qq = ctx.Set<AdminAuditLog>().AsNoTracking();
            if (q.AfterCreatedAt is { } ts && q.AfterId is { } aid)
                qq = qq.Where(r => r.CreatedAt < ts || (r.CreatedAt == ts && r.Id < aid));
            if (!string.IsNullOrEmpty(q.Action)) qq = qq.Where(r => r.Action == q.Action);
            var rows = await qq.OrderByDescending(r => r.CreatedAt).ThenByDescending(r => r.Id)
                .Take(page + 1).ToListAsync(ct);
            var hasMore = rows.Count > page;
            if (hasMore) rows.RemoveAt(page);
            return Results.Ok(new { items = rows, hasMore });
        }

        private static async Task<IResult> GetMatchHistoryAsync(
            Guid playerId, int pageSize, GameKitDbContext ctx, CancellationToken ct)
        {
            // Direct EF query against SessionParticipant joined to Player.
            // Reuse GDPR export pattern (Phase 1 Core GdprDeleteService reference point).
            pageSize = Math.Clamp(pageSize, 1, 50);
            var rows = await ctx.Set<SessionParticipant>().AsNoTracking()
                .Where(p => p.PlayerId == playerId && p.Session!.State == GameSessionState.Completed)
                .OrderByDescending(p => p.Session!.EndedAt)
                .Take(pageSize)
                .Select(p => new MatchHistoryRow(
                    p.SessionId, p.Session!.LadderId, p.Team, p.Result,
                    p.RatingBefore, p.RatingAfter, p.Delta, p.Session.EndedAt))
                .ToListAsync(ct);
            return Results.Ok(rows);
        }

        private static async Task<IResult> GetHealthAsync(IHealthProbeService svc, CancellationToken ct)
            => Results.Ok(await svc.GetReportAsync(ct));

        private static Guid GetAdminId(HttpContext http)
        {
            var nameId = http.User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
            return Guid.TryParse(nameId, out var id) ? id : throw new UnauthorizedAccessException();
        }

        private sealed record AuditQuery(DateTimeOffset? AfterCreatedAt, Guid? AfterId, string? Action, int? PageSize);
        public sealed record MatchHistoryRow(
            Guid SessionId, Guid? LadderId, int Team, int? Result,
            double? RatingBefore, double? RatingAfter, double? Delta, DateTimeOffset? CompletedAt);
    }
