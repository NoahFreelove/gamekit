  ```csharp
  // SPDX-License-Identifier: GPL-3.0-or-later
  // Copyright (c) 2026 GameKit contributors

  using System.Linq;
  using System.Threading.Tasks;
  using FluentValidation;
  using Microsoft.AspNetCore.Http;
  using Microsoft.Extensions.DependencyInjection;

  namespace GameKit.Auth.Http.EndpointFilters;

  /// &lt;summary&gt;
  /// Generic &lt;see cref="IEndpointFilter"/&gt; that resolves &lt;c&gt;IValidator&amp;lt;TRequest&amp;gt;&lt;/c&gt; from DI and
  /// runs it against the first request-body argument found in the endpoint invocation. Returns
  /// &lt;see cref="Results.ValidationProblem"/&gt; on failure. AUTH + FluentValidation 12 + minimal APIs
  /// — no auto-MVC wiring (CLAUDE.md).
  /// &lt;/summary&gt;
  public sealed class ValidationEndpointFilter&lt;TRequest&gt; : IEndpointFilter where TRequest : class
  {
      /// &lt;inheritdoc /&gt;
      public async ValueTask&lt;object?&gt; InvokeAsync(
          EndpointFilterInvocationContext ctx, EndpointFilterDelegate next)
      {
          var req = ctx.Arguments.OfType&lt;TRequest&gt;().FirstOrDefault();
          if (req is null) return await next(ctx);

          var validator = ctx.HttpContext.RequestServices.GetService&lt;IValidator&lt;TRequest&gt;&gt;();
          if (validator is null) return await next(ctx);

          var result = await validator.ValidateAsync(req, ctx.HttpContext.RequestAborted).ConfigureAwait(false);
          return result.IsValid
              ? await next(ctx)
              : Results.ValidationProblem(result.ToDictionary());
      }
  }
  ```

  (e) `src/GameKit.Auth/Http/RateLimiting/AuthRateLimitRegistrations.cs` — RESEARCH §8.7 verbatim:

  ```csharp
  // SPDX-License-Identifier: GPL-3.0-or-later
  // Copyright (c) 2026 GameKit contributors

  using System;
  using System.Globalization;
  using System.Threading.RateLimiting;
  using GameKit.Core.RateLimiting;
  using Microsoft.AspNetCore.Http;
  using Microsoft.AspNetCore.RateLimiting;
  using Microsoft.Extensions.DependencyInjection;

  namespace GameKit.Auth.Http.RateLimiting;

  /// &lt;summary&gt;
  /// Registers fixed-window rate-limit policies under the Phase-1 named-policy constants.
  /// Partition key = IP + X-GameKit-Device (defends against fingerprint-spray DoS; RESEARCH Pitfall D).
  /// &lt;/summary&gt;
  public static class AuthRateLimitRegistrations
  {
      /// &lt;summary&gt;Adds the three Auth rate-limit policies to the service collection.&lt;/summary&gt;
      public static IServiceCollection AddAuthRateLimits(this IServiceCollection services, IGameKitRateLimitPolicies names)
      {
          services.AddRateLimiter(opt =&gt;
          {
              opt.RejectionStatusCode = StatusCodes.Status429TooManyRequests;
              opt.OnRejected = async (ctx, ct) =&gt;
              {
                  if (ctx.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retryAfter))
                      ctx.HttpContext.Response.Headers.RetryAfter =
                          ((int)retryAfter.TotalSeconds).ToString(CultureInfo.InvariantCulture);
                  ctx.HttpContext.Response.StatusCode = StatusCodes.Status429TooManyRequests;
                  ctx.HttpContext.Response.ContentType = "application/problem+json";
                  await ctx.HttpContext.Response.WriteAsJsonAsync(new
                  {
                      type = "https://gamekit.dev/errors/rate-limit",
                      title = "Too Many Requests",
                      status = 429,
                  }, ct);
              };

              AddPolicy(opt, names.AuthLogin,    permit: 10, window: TimeSpan.FromMinutes(1));
              AddPolicy(opt, names.AuthRefresh,  permit: 60, window: TimeSpan.FromMinutes(1));
              AddPolicy(opt, names.AuthRegister, permit:  5, window: TimeSpan.FromMinutes(1));
          });
          return services;
      }

      private static void AddPolicy(RateLimiterOptions opt, string name, int permit, TimeSpan window)
      {
          opt.AddPolicy(name, httpContext =&gt;
          {
              var ip = httpContext.Connection.RemoteIpAddress?.ToString() ?? "unknown";
              var fp = httpContext.Request.Headers["X-GameKit-Device"].ToString();
              var partitionKey = string.IsNullOrEmpty(fp) ? ip : $"{ip}:{fp}";

              return RateLimitPartition.GetFixedWindowLimiter(
                  partitionKey: partitionKey,
                  factory: _ =&gt; new FixedWindowRateLimiterOptions
                  {
                      PermitLimit = permit,
                      Window = window,
                      QueueLimit = 0,
                      AutoReplenishment = true,
                  });
          });
      }
  }
  ```

  ```csharp
      public static RouteGroupBuilder MapAuthEndpoints(this IEndpointRouteBuilder routes, IGameKitRateLimitPolicies policies)
      {

          grp.MapPost("/logout", LogoutAsync)

          grp.MapGet("/callback/{provider}", CallbackAsync);        // browser GET

              .AddEndpointFilter&lt;ValidationEndpointFilter&lt;LinkRequest&gt;&gt;()
              .RequireAuthorization();

      private static async Task&lt;IResult&gt; LoginAsync(
          string provider,
          LoginRequest req,
          HttpContext http,
          IServiceProvider sp,
          CancellationToken ct)
      {
          var providers = sp.GetServices&lt;IOAuthProvider&gt;().Where(p =&gt; p.Provider == provider).ToList();
          var impl = providers.FirstOrDefault();
          if (impl is null) return Results.BadRequest(new AuthErrorResponse("unknown_provider", provider));

          var fingerprint = GetFingerprint(http);
          var result = await impl.CompleteLoginAsync(
              externalId: req.Username ?? string.Empty,
              displayName: req.Password,   // password provider convention; other providers ignore
              avatarUrl: null,
              fingerprint: fingerprint,
              cancellationToken: ct).ConfigureAwait(false);

          return result.Success
              ? Results.Ok(new TokenResponse(result.Tokens!.AccessJwt, result.Tokens!.RawRefresh))
              : Results.Json(new AuthErrorResponse(result.ErrorCode ?? "invalid_credentials", provider), statusCode: StatusCodes.Status401Unauthorized);
      }

      private static async Task&lt;IResult&gt; RefreshAsync(
          RefreshRequest req,
          HttpContext http,
          IRefreshTokenService svc,
          CancellationToken ct)
      {
          var fingerprint = GetFingerprint(http);
          try
          {
              var pair = await svc.RotateAsync(req.RefreshToken, fingerprint, ct).ConfigureAwait(false);
              return Results.Ok(new TokenResponse(pair.AccessJwt, pair.RawRefresh));
          }
          catch (UnauthorizedException ux)
          {
              return Results.Json(new AuthErrorResponse(ux.Code), statusCode: StatusCodes.Status401Unauthorized);
          }
      }

      private static async Task&lt;IResult&gt; RegisterAsync(
          RegisterRequest req,
          HttpContext http,
          PasswordOAuthProvider passwordProvider,
          IGuestUpgradeService upgrade,
          CancellationToken ct)
      {
          var fingerprint = GetFingerprint(http);

          // D-12: if caller is authenticated as a guest, upgrade in place.
          if (http.User.Identity?.IsAuthenticated == true)
          {
              var sub = http.User.FindFirst("sub")?.Value ?? http.User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
              var isGuest = http.User.FindFirst("is_guest")?.Value == "true";
              if (sub is not null &amp;&amp; Guid.TryParse(sub, out var playerId) &amp;&amp; isGuest)
              {
                  try
                  {
                      var tokens = await upgrade.UpgradeToPasswordAsync(playerId, req.Username, req.Password, fingerprint, ct).ConfigureAwait(false);
                      return Results.Ok(new TokenResponse(tokens.AccessJwt, tokens.RawRefresh));
                  }
                  catch (UsernameAlreadyTakenException)
                  {
                      return Results.Json(new AuthErrorResponse("username_taken"), statusCode: StatusCodes.Status409Conflict);
                  }
              }
          }

          var result = await passwordProvider.RegisterAsync(req.Username, req.Password, req.DisplayName, fingerprint, ct).ConfigureAwait(false);
          if (!result.Success)
          {
              var status = result.ErrorCode == "username_taken" ? StatusCodes.Status409Conflict : StatusCodes.Status400BadRequest;
              return Results.Json(new AuthErrorResponse(result.ErrorCode ?? "bad_request"), statusCode: status);
          }
          return Results.Ok(new TokenResponse(result.Tokens!.AccessJwt, result.Tokens!.RawRefresh));
      }

      private static async Task&lt;IResult&gt; LogoutAsync(LogoutRequest req, IRefreshTokenService svc, CancellationToken ct)
      {
          await svc.RevokeFamilyAsync(req.RefreshToken, "manual_logout", ct).ConfigureAwait(false);
          return Results.Ok();
      }

      private static async Task&lt;IResult&gt; LogoutAllAsync(HttpContext http, IRefreshTokenService svc, CancellationToken ct)
      {
          var sub = http.User.FindFirst("sub")?.Value;
          if (sub is null || !Guid.TryParse(sub, out var playerId))
              return Results.Unauthorized();
          await svc.RevokeAllForPlayerAsync(playerId, "logout_all", ct).ConfigureAwait(false);
          return Results.Ok();
      }

      private static IResult MeAsync(HttpContext http)
      {
          var sub = http.User.FindFirst("sub")?.Value;
          var isGuest = http.User.FindFirst("is_guest")?.Value;
          var provider = http.User.FindFirst("provider")?.Value;
          if (sub is null) return Results.Unauthorized();
          return Results.Ok(new { player_id = sub, is_guest = isGuest == "true", provider });
      }

      private static IResult ChallengeAsync(
          string provider,
          HttpContext http,
          GameKitAuthOptions opts)
      {
          if (provider == "steam")
          {
              // Compose openid.* query params and 302.
              var realm = opts.Steam.Realm;
              if (string.IsNullOrEmpty(realm))
                  realm = $"{http.Request.Scheme}://{http.Request.Host}/";
              var returnTo = new Uri(new Uri(realm), opts.Steam.CallbackPath).ToString();
              var qs = new Dictionary&lt;string, string&gt;
              {
                  ["openid.ns"] = "http://specs.openid.net/auth/2.0",
                  ["openid.mode"] = "checkid_setup",
                  ["openid.return_to"] = returnTo,
                  ["openid.realm"] = realm,
                  ["openid.identity"] = "http://specs.openid.net/auth/2.0/identifier_select",
                  ["openid.claimed_id"] = "http://specs.openid.net/auth/2.0/identifier_select",
              };
              var query = string.Join("&amp;", qs.Select(kv =&gt; $"{Uri.EscapeDataString(kv.Key)}={Uri.EscapeDataString(kv.Value)}"));
              return Results.Redirect($"{opts.Steam.OpenIdEndpoint}?{query}");
          }
          if (provider == "discord")
          {
              return Results.Challenge(new AuthenticationProperties { RedirectUri = opts.Discord.CallbackPath }, new[] { "Discord" });
          }
          return Results.BadRequest(new AuthErrorResponse("unknown_provider", provider));
      }

      private static async Task&lt;IResult&gt; CallbackAsync(
          string provider,
          HttpContext http,
          IServiceProvider sp,
          CancellationToken ct)
      {
          if (provider == "steam")
          {
              var verifier = sp.GetRequiredService&lt;SteamOpenIdVerifier&gt;();
              var v = await verifier.VerifyAsync(http.Request.Query, ct).ConfigureAwait(false);
              if (!v.IsValid)
                  return Results.BadRequest(new AuthErrorResponse("invalid_assertion", "steam"));

              var fingerprint = GetFingerprint(http);
              var steamProvider = sp.GetServices&lt;IOAuthProvider&gt;().First(p =&gt; p.Provider == "steam");
              var result = await steamProvider.CompleteLoginAsync(v.SteamId64!, null, null, fingerprint, ct).ConfigureAwait(false);
              return result.Success
                  ? Results.Ok(new TokenResponse(result.Tokens!.AccessJwt, result.Tokens!.RawRefresh))
                  : Results.Json(new AuthErrorResponse("login_failed", "steam"), statusCode: 401);
          }

          if (provider == "discord")
          {
              // Plan 02-05's Events.OnCreatingTicket stashed tokens into Properties.Items.
              var authResult = await http.AuthenticateAsync("Discord").ConfigureAwait(false);
              if (!authResult.Succeeded || authResult.Properties is null)
                  return Results.Unauthorized();

              var access = authResult.Properties.Items.TryGetValue("gamekit.access_jwt", out var a) ? a : null;
              var refresh = authResult.Properties.Items.TryGetValue("gamekit.refresh_raw", out var r) ? r : null;
              if (access is null) return Results.Unauthorized();
              return Results.Ok(new TokenResponse(access, refresh));
          }

          return Results.BadRequest(new AuthErrorResponse("unknown_provider", provider));
      }

      private static async Task&lt;IResult&gt; LinkAsync(
          string provider,
          LinkRequest req,
          HttpContext http,
          IIdentityLinker linker,
          SteamOpenIdVerifier steamVerifier,
          CancellationToken ct)
      {
          var sub = http.User.FindFirst("sub")?.Value;
          if (sub is null || !Guid.TryParse(sub, out var playerId))
              return Results.Unauthorized();

          string externalId;
          if (provider == "steam")
          {
              // Either body.ExternalId (SteamID64 already verified by client flow) OR run the verifier on the query params.
              if (!string.IsNullOrEmpty(req.ExternalId))
              {
                  externalId = req.ExternalId!;
              }
              else
              {
                  var v = await steamVerifier.VerifyAsync(http.Request.Query, ct).ConfigureAwait(false);
                  if (!v.IsValid) return Results.BadRequest(new AuthErrorResponse("invalid_assertion", "steam"));
                  externalId = v.SteamId64!;
              }
          }
          else if (provider == "discord")
          {
              if (string.IsNullOrEmpty(req.ExternalId))
                  return Results.BadRequest(new AuthErrorResponse("external_id_required", "discord"));
              externalId = req.ExternalId!;
          }
          else
          {
              return Results.BadRequest(new AuthErrorResponse("unknown_provider", provider));
          }

          var result = await linker.LinkAsync(playerId, provider, externalId, ct).ConfigureAwait(false);
          return result.Kind switch
          {
              LinkResultKind.Linked or LinkResultKind.AlreadyLinkedToSelf =&gt; Results.Ok(),
              LinkResultKind.AlreadyLinkedToOtherPlayer =&gt;
                  Results.Json(new AuthErrorResponse("identity_already_linked", provider, result.ExternalIdHash), statusCode: StatusCodes.Status409Conflict),
              _ =&gt; Results.Problem(statusCode: StatusCodes.Status500InternalServerError),
          };
      }

      private static string? GetFingerprint(HttpContext http)
      {
          var raw = http.Request.Headers["X-GameKit-Device"].ToString();
          return string.IsNullOrEmpty(raw) ? null : raw;
      }
  }
  ```
