  2. (Optional) Set real Discord OAuth creds in `appsettings.Development.json` if you
     want to exercise the Discord flow:

     ```json
     "Discord": { "ClientId": "xxx", "ClientSecret": "yyy" }
     ```

     The guest and password providers work without any external credentials.

  ### Client token storage (localStorage) — security note

  The HTML client stores JWTs in `localStorage` so the browser reload UX works.
  **This is XSS-vulnerable** — if any script executing on the page can read
  `localStorage`, the access + refresh tokens are stolen. Acceptable trade-off for a
  demo; production apps should use one of:

  - HTTP-only `Secure` + `SameSite=Strict` cookies set server-side (changes the API
    contract — GameKit.Auth returns tokens in the response body for portability).
  - A native token cache (Android/iOS keychain, Electron preload script).
  - A Service-Worker-mediated split: refresh in a background context, access-only in
    the page's JS.

  ### Signing-key hygiene

  `appsettings.Development.json` references `samples/TicTacToeDuel/keys/dev-priv.pem`.

  - **Never** commit a real private key to git. `.gitignore` in `samples/TicTacToeDuel/keys/` excludes `*.pem`.
  - In production, the private key file **must** be mode `0600` and owned by the
    process user. Anyone who reads this file can forge tokens indistinguishable from
    the server's.
  - Rotate: generate a new `Kid`, switch issuance, keep the old public key in the
    validator for 30 days (refresh-token lifetime) so existing sessions still verify.

  ### Customizing the egress allow-list

  `GameKit.Auth` only talks to hosts on
  `GameKitAuthOptions.AllowedProviderHosts`. Defaults cover Steam + Discord; if you
  proxy OAuth through another host, add it:

  ```csharp
  .AddAuth(auth =&gt;
  {
      // ... existing options ...
      auth.AllowedProviderHosts.Add("id.internal.example.com");
  });
  ```

  An off-list outbound call throws `EgressViolationException` — loud failure, not
  silent leak.

      localStorage.removeItem(KEY_REFRESH);
  }

  function getAccess() { return localStorage.getItem(KEY_ACCESS); }
  function getRefresh() { return localStorage.getItem(KEY_REFRESH); }

  // Client-side JWT payload decoder (display only — server still validates signatures).
  function decodeJwtPayload(jwt) {
      const parts = jwt.split(".");
      if (parts.length !== 3) return null;
      const payload = JSON.parse(atob(parts[1].replace(/-/g, "+").replace(/_/g, "/")));
      return payload;
  }

  async function refreshTokens() {
      const refresh = getRefresh();
      if (!refresh) return false;
      const resp = await fetch("/auth/refresh", {
          method: "POST",
          headers: { "Content-Type": "application/json", "X-GameKit-Device": getOrCreateDeviceId() },
          body: JSON.stringify({ refreshToken: refresh })
      });
      if (!resp.ok) return false;
      const body = await resp.json();
      // Server may omit refreshToken when returning an idempotent replay inside the grace window.
      setTokens(body.accessToken, body.refreshToken ?? refresh);
      return true;
  }

  async function gkFetch(path, init) {
      init = init || {};
      init.headers = init.headers || {};
      init.headers["X-GameKit-Device"] = getOrCreateDeviceId();
      const access = getAccess();
      if (access) init.headers["Authorization"] = "Bearer " + access;

      let resp = await fetch(path, init);
      if (resp.status === 401 &amp;&amp; getRefresh()) {
          const ok = await refreshTokens();
          if (ok) {
              init.headers["Authorization"] = "Bearer " + getAccess();
              resp = await fetch(path, init);
          }
          if (resp.status === 401) {
              clearTokens();
              renderAuthUi();
              throw new Error("unauthorized");
          }
      }
      return resp;
  }

  function renderAuthUi() {
      const access = getAccess();
      const authPanel = document.getElementById("auth-panel");
      const gamePanel = document.getElementById("game-panel");
      if (access) {
          const payload = decodeJwtPayload(access);
          document.getElementById("me-display").textContent = payload?.sub ?? "(unknown)";
          document.getElementById("me-isguest").textContent = payload?.is_guest ? "true" : "false";
          authPanel.classList.add("hidden");
          gamePanel.classList.remove("hidden");
      } else {
          authPanel.classList.remove("hidden");
          gamePanel.classList.add("hidden");
      }
  }

  async function doGuestLogin() {
      const resp = await fetch("/auth/login/guest", {
          method: "POST",
          headers: { "Content-Type": "application/json", "X-GameKit-Device": getOrCreateDeviceId() },
          body: JSON.stringify({})
      });
      if (!resp.ok) { document.getElementById("auth-error").textContent = "guest login failed"; return; }
      const body = await resp.json();
      setTokens(body.accessToken, body.refreshToken);
      renderAuthUi();
  }

  async function doRegister() {
      const username = document.getElementById("username").value;
      const password = document.getElementById("password").value;
      // If already authenticated as guest, /auth/register upgrades in place (D-12).
      const init = {
          method: "POST",
          headers: { "Content-Type": "application/json", "X-GameKit-Device": getOrCreateDeviceId() },
          body: JSON.stringify({ username, password })
      };
      const access = getAccess();
      if (access) init.headers["Authorization"] = "Bearer " + access;
      const resp = await fetch("/auth/register", init);
      if (!resp.ok) { document.getElementById("auth-error").textContent = "register failed (" + resp.status + ")"; return; }
      const body = await resp.json();
      setTokens(body.accessToken, body.refreshToken);
      renderAuthUi();
  }

  async function doLogin() {
      const username = document.getElementById("username").value;
      const password = document.getElementById("password").value;
      const resp = await fetch("/auth/login/password", {
          method: "POST",
          headers: { "Content-Type": "application/json", "X-GameKit-Device": getOrCreateDeviceId() },
          body: JSON.stringify({ username, password })
      });
      if (!resp.ok) { document.getElementById("auth-error").textContent = "login failed"; return; }
      const body = await resp.json();
      setTokens(body.accessToken, body.refreshToken);
      renderAuthUi();
  }

  async function doLogout() {
      const refresh = getRefresh();
      if (refresh) {
          await fetch("/auth/logout", {
              method: "POST",
              headers: { "Content-Type": "application/json", "X-GameKit-Device": getOrCreateDeviceId(), "Authorization": "Bearer " + getAccess() },
              body: JSON.stringify({ refreshToken: refresh })
          });
      }
      clearTokens();
      renderAuthUi();
  }

  document.getElementById("play-guest").addEventListener("click", doGuestLogin);
  document.getElementById("btn-register").addEventListener("click", doRegister);
  document.getElementById("btn-login").addEventListener("click", doLogin);
  document.getElementById("btn-logout").addEventListener("click", doLogout);

  getOrCreateDeviceId();
  renderAuthUi();

  // Phase-1 game UI: the demo board + move endpoints stay. Hook them up to gkFetch(...) for auth-aware requests.
  // (Keep the existing game logic from the Phase-1 index.html — only the auth shell changed.)
