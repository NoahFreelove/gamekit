        private readonly IClock _clock;
        private readonly IIdGenerator _ids;
        public AdminAuditWriter(GameKitDbContext ctx, IClock clock, IIdGenerator ids)

            var row = new AdminAuditLog

                CreatedAt = _clock.UtcNow,

            await _ctx.SaveChangesAsync(ct).ConfigureAwait(false);
        }
    }

    ```csharp
    // Public enum + struct for classification (tested in PlayerSearchInputDetectionTests)
    public enum SearchMode { None, Id, Identity, DisplayName }

    public readonly record struct PlayerSearchClassification(
        SearchMode Mode, Guid Id, string Provider, string ExternalId, string DisplayName);

    public interface IPlayerSearchService
    {
        Task<PaginatedResult<PlayerRow>> SearchAsync(string query, Guid? afterId, int pageSize, CancellationToken ct);
    }

    internal sealed class PlayerSearchService : IPlayerSearchService
    {
        private readonly GameKitDbContext _ctx;
        public PlayerSearchService(GameKitDbContext ctx) { _ctx = ctx; }

        public static PlayerSearchClassification ClassifyInput(string? raw)
        {
            if (string.IsNullOrWhiteSpace(raw)) return new(SearchMode.None, default, "", "", "");
            var q = raw.Trim();
            if (Guid.TryParse(q, out var id)) return new(SearchMode.Id, id, "", "", "");
            var colon = q.IndexOf(':');
            if (colon > 0 && colon < q.Length - 1)
            {
                var p = q[..colon]; var ext = q[(colon+1)..];
                if (p.Length is >= 2 and <= 32 && ext.Length is >= 1 and <= 256)
                    return new(SearchMode.Identity, Guid.Empty, p, ext, "");
            }
            return new(SearchMode.DisplayName, Guid.Empty, "", "", q);
        }

        public async Task<PaginatedResult<PlayerRow>> SearchAsync(string query, Guid? afterId, int pageSize, CancellationToken ct)
        {
            var c = ClassifyInput(query);
            pageSize = Math.Clamp(pageSize, 1, 50);
            switch (c.Mode)
            {
                case SearchMode.None: return PaginatedResult<PlayerRow>.Empty;
                case SearchMode.Id:
                {
                    var p = await _ctx.Set<Player>().AsNoTracking()
                        .Where(x => x.Id == c.Id)
                        .Select(x => new PlayerRow(x.Id, x.DisplayName, x.CreatedAt, x.IsBanned))
                        .FirstOrDefaultAsync(ct).ConfigureAwait(false);
                    return p is null ? PaginatedResult<PlayerRow>.Empty
                        : new PaginatedResult<PlayerRow>(new[]{p}, null, false);
                }
                case SearchMode.Identity:
                {
                    var row = await (from i in _ctx.Set<PlayerIdentity>().AsNoTracking()
                                     where i.Provider == c.Provider && i.ExternalId == c.ExternalId
                                     join p in _ctx.Set<Player>().AsNoTracking() on i.PlayerId equals p.Id
                                     select new PlayerRow(p.Id, p.DisplayName, p.CreatedAt, p.IsBanned))
                                     .FirstOrDefaultAsync(ct).ConfigureAwait(false);
                    return row is null ? PaginatedResult<PlayerRow>.Empty
                        : new PaginatedResult<PlayerRow>(new[]{row}, null, false);
                }
                default: // DisplayName — keyset prefix
                {
                    var q = _ctx.Set<Player>().AsNoTracking()
                        .Where(p => EF.Functions.ILike(p.DisplayName, c.DisplayName + "%"));
                    if (afterId is not null) q = q.Where(p => p.Id < afterId.Value);
                    var rows = await q.OrderByDescending(p => p.Id).Take(pageSize + 1)
                        .Select(p => new PlayerRow(p.Id, p.DisplayName, p.CreatedAt, p.IsBanned))
                        .ToListAsync(ct).ConfigureAwait(false);
                    var hasMore = rows.Count > pageSize;
                    if (hasMore) rows.RemoveAt(pageSize);
                    return new PaginatedResult<PlayerRow>(rows, hasMore ? rows[^1].Id.ToString() : null, hasMore);
                }
            }
        }
    }
    ```

    public interface IAdminAuthService
    {
        /// <summary>Returns admin id + role on success, null on failure (wrong password or unknown user).</summary>
        Task<(Guid AdminId, string Role)?> VerifyPasswordAsync(string username, string password, CancellationToken ct);
    }

    internal sealed class AdminAuthService : IAdminAuthService
    {
        // Canned BCrypt hash (work-factor 12) for the literal placeholder "x" — NEVER a real admin password.
        // Identical pattern to Phase-2 T-02-16 mitigation (PasswordOAuthProvider.DummyHash). The literal below is
        // exactly 60 characters ($2a$12$ + 22-char salt + 31-char ciphertext), verified BCrypt.Net-Next compatible.
        // Paste VERBATIM — do not regenerate; deterministic source of the dummy-hash must survive CI re-runs.
        private const string DummyHash = "$2a$12$KIXfJpcJf8yNrIxS.GkJd.UgL0VtZa7lF5E5qKTQDHQwuVZmDoYra";
        private readonly GameKitDbContext _ctx;
        private readonly IPasswordHasher _hasher;
        private readonly IAdminAuditWriter _audit;
        private readonly IClock _clock;
        public AdminAuthService(GameKitDbContext ctx, IPasswordHasher hasher, IAdminAuditWriter audit, IClock clock)
        { _ctx = ctx; _hasher = hasher; _audit = audit; _clock = clock; }

        public async Task<(Guid,string)?> VerifyPasswordAsync(string username, string password, CancellationToken ct)
        {
            var admin = await _ctx.Set<AdminUser>().AsTracking()
                .FirstOrDefaultAsync(a => a.Username == username, ct).ConfigureAwait(false);
            if (admin is null)
            {
                _hasher.Verify(password, DummyHash); // timing parity
                return null;
            }
            if (admin.LockedUntil is { } lu && lu > _clock.UtcNow)
            {
                await _audit.WriteAsync(AdminAuditActions.SessionLoginFailure, "admin", admin.Id, admin.Id, null, new { reason = "locked" }, null, ct).ConfigureAwait(false);
                return null;
            }
            if (!_hasher.Verify(password, admin.PasswordHash))
            {
                admin.FailedLoginCount += 1;
                await _audit.WriteAsync(AdminAuditActions.SessionLoginFailure, "admin", admin.Id, admin.Id, null, new { failed_count = admin.FailedLoginCount }, null, ct).ConfigureAwait(false);
                return null;
            }
            admin.FailedLoginCount = 0;

    // src/GameKit.Admin.UI/Services/ErrorRateRingBuffer.cs
    // SPDX-License-Identifier: GPL-3.0-or-later
    // Copyright (c) 2026 GameKit contributors
    using System;
    using System.Threading;
    using GameKit.Core.Abstractions;  // IClock

    namespace GameKit.Admin.UI.Services;

    /// <summary>
    /// Thread-safe rolling-window error counter. Buckets are fixed-width (configurable, default 1s); the
    /// total window equals <c>GameKitAdminOptions.Panel.HealthErrorRateWindow</c> (default 5m). Increments
    /// are O(1) lock-free (Interlocked.Increment); reads (<see cref="RecentErrorCount"/>) rotate and sum.
    /// Older buckets naturally decay as newer ticks overwrite them. See RESEARCH §Health panel (lines 894-938).
    /// </summary>
    public sealed class ErrorRateRingBuffer
    {
        private readonly int[] _buckets;          // one counter per bucket
        private readonly long _bucketTicks;       // fixed bucket width (ticks)
        private readonly IClock _clock;
        private long _headBucketStartTicks;       // start tick of the bucket at _headIndex
        private int _headIndex;                   // index of the currently-live bucket
        private readonly object _rotateGate = new();

        /// <summary>Constructs the buffer from options + clock.</summary>
        public ErrorRateRingBuffer(GameKitAdminOptions opts, IClock clock)
        {
            ArgumentNullException.ThrowIfNull(opts);
            ArgumentNullException.ThrowIfNull(clock);
            var window = opts.Panel.HealthErrorRateWindow;
            var bucket = opts.Panel.HealthErrorRateBucketSize;
            if (bucket <= TimeSpan.Zero) throw new ArgumentException("bucket size must be > 0", nameof(opts));
            if (window < bucket) throw new ArgumentException("window must be >= bucket size", nameof(opts));
            var count = (int)Math.Max(1, window.Ticks / bucket.Ticks);
            _buckets = new int[count];
            _bucketTicks = bucket.Ticks;
            _clock = clock;
            _headBucketStartTicks = NormalizeToBucket(_clock.UtcNow.UtcTicks);
            _headIndex = 0;
        }

        /// <summary>Records one error event at the current time. O(1), lock-free on the hot path.</summary>
        public void IncrementError()
        {
            AdvanceIfNeeded();
            Interlocked.Increment(ref _buckets[_headIndex]);
        }

        /// <summary>Returns the sum of all buckets in the current rolling window.</summary>
        public int RecentErrorCount()
        {
            AdvanceIfNeeded();
            var sum = 0;
            for (var i = 0; i < _buckets.Length; i++) sum += Volatile.Read(ref _buckets[i]);
            return sum;
        }

        private long NormalizeToBucket(long ticks) => ticks - (ticks % _bucketTicks);

        private void AdvanceIfNeeded()
        {
            var nowTicks = _clock.UtcNow.UtcTicks;
            var elapsed = nowTicks - _headBucketStartTicks;
            if (elapsed < _bucketTicks) return;  // still in the same bucket — hot path exits here

            lock (_rotateGate)
            {
                // Re-check under lock (double-checked locking pattern).
                elapsed = _clock.UtcNow.UtcTicks - _headBucketStartTicks;
                if (elapsed < _bucketTicks) return;

                var stepsRaw = elapsed / _bucketTicks;
                var steps = (int)Math.Min(stepsRaw, _buckets.Length);
                for (var i = 0; i < steps; i++)
                {
                    _headIndex = (_headIndex + 1) % _buckets.Length;
                    _buckets[_headIndex] = 0;  // zero the bucket we're rotating onto
                }
                _headBucketStartTicks += stepsRaw * _bucketTicks;
            }
        }
    }

    public static class AdminBuilderExtensions
    {
        public static IGameKitBuilder AddGameKitAdmin(
            this IGameKitBuilder builder,
            Action<GameKitAdminOptions>? configure = null)
        {
            ArgumentNullException.ThrowIfNull(builder);
            var opts = new GameKitAdminOptions();
            configure?.Invoke(opts);
            ValidateAdminOptions(opts);
            builder.Services.AddSingleton(opts);

            // 1. Admin model-builder extension
            builder.Services.TryAddEnumerable(
                ServiceDescriptor.Singleton<IModelBuilderExtension, AdminModelBuilderExtension>());
            // 2. Migration hosted service — MUST precede SuperadminGate
            builder.Services.AddHostedService<AdminMigrationHostedService>();
            // 3. Superadmin gate hosted service (D-04)
            builder.Services.AddHostedService<SuperadminGateHostedService>();

            // 4. Cookie auth scheme — distinct from Bearer.
            //    W4: preserve Phase 2 JWT Bearer as the DEFAULT scheme (AddAuthentication(JwtBearerDefaults.AuthenticationScheme))
            //    and register the admin cookie as a NAMED scheme only. This ensures existing player-JWT endpoints
            //    (e.g. /auth/me) continue to authenticate with Bearer when GameKit.Admin.UI is added. Admin endpoints
            //    opt in to the named cookie scheme explicitly via the authorization policies below.
            builder.Services.AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
                .AddCookie(AdminAuthenticationSchemeConstants.Scheme, o =>
                {
                    o.Cookie.Name = opts.Cookie.Name;
                    o.Cookie.HttpOnly = true;
                    o.Cookie.SecurePolicy = CookieSecurePolicy.Always;
                    o.Cookie.SameSite = SameSiteMode.Lax;
                    o.ExpireTimeSpan = opts.Cookie.ExpireTimeSpan;
                    o.SlidingExpiration = opts.Cookie.SlidingExpiration;
                    // Blazor Razor pages are fixed at /admin/* regardless of MountPath (which scopes only
                    // the HTTP API prefix — see GameKitAdminOptions.MountPath XML doc). Cookie redirect
                    // targets must reference the Blazor login page at its real route.
                    o.LoginPath = "/admin/login";
                    o.LogoutPath = "/admin/logout";
                    o.EventsType = typeof(AdminCookieEvents);
                });
            builder.Services.AddScoped<AdminCookieEvents>();

            // 5. Authorization policies — pin scheme
            builder.Services.AddAuthorization(ao =>
            {
                ao.AddPolicy(AdminPolicies.Admin, p => p
                    .AddAuthenticationSchemes(AdminAuthenticationSchemeConstants.Scheme)
                    .RequireAuthenticatedUser()
                    .RequireRole(AdminRoles.Admin, AdminRoles.Superadmin));
                ao.AddPolicy(AdminPolicies.Superadmin, p => p
                    .AddAuthenticationSchemes(AdminAuthenticationSchemeConstants.Scheme)
                    .RequireAuthenticatedUser()
                    .RequireRole(AdminRoles.Superadmin));
            });

            // 6. Services
            builder.Services.AddScoped<IAdminAuditWriter, AdminAuditWriter>();
            builder.Services.AddScoped<IAdminAuthService, AdminAuthService>();
            builder.Services.AddScoped<IPlayerSearchService, PlayerSearchService>();
            builder.Services.AddScoped<IPlayerBanService, PlayerBanService>();
            builder.Services.AddScoped<IAdminUserService, AdminUserService>();
            builder.Services.AddScoped<IHealthProbeService, HealthProbeService>();
            builder.Services.AddSingleton<ErrorRateRingBuffer>();
            builder.Services.AddSingleton<ILoggerProvider, LogErrorCounter>();

            // 7. Rate limiter
            builder.Services.AddAdminRateLimits();

            // 8. Antiforgery (header name per D-16)
            builder.Services.AddAntiforgery(o =>
            {
                o.HeaderName = AdminAuthenticationSchemeConstants.CsrfHeaderName;
                o.Cookie.Name = AdminAuthenticationSchemeConstants.CsrfCookieName;
                o.Cookie.HttpOnly = false; // JS reads and echoes via header
                o.Cookie.SameSite = SameSiteMode.Lax;
                o.Cookie.SecurePolicy = CookieSecurePolicy.Always;
            });

            // 9. Blazor Server primitives
            builder.Services.AddRazorComponents().AddInteractiveServerComponents();

            // 10. MudBlazor
            builder.Services.AddMudServices();

            // 11. IHttpContextAccessor (required by App.razor to read CSP nonce)
            builder.Services.AddHttpContextAccessor();

            // ^ If plan 03-07 has not shipped yet, add the validators that exist; others land with 03-07.

            return builder;

        internal static void ValidateAdminOptions(GameKitAdminOptions opts)
        {

            if (opts.Cookie.ExpireTimeSpan <= TimeSpan.Zero)
                throw new ArgumentException("GameKitAdminOptions.Cookie.ExpireTimeSpan must be > 0.", nameof(opts));
        }
