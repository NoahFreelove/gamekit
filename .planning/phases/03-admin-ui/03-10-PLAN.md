    ```csharp
    // SPDX-License-Identifier: GPL-3.0-or-later
    // Copyright (c) 2026 GameKit contributors
    using System;
    using System.Security.Cryptography;
    using System.Text;
    using System.Threading;
    using System.Threading.Tasks;
    using GameKit.Auth.Providers;
    using GameKit.Core.Data;
    using GameKit.Core.Entities;
    using Microsoft.EntityFrameworkCore;

    namespace GameKit.Auth.Services;

    /// <summary>
    /// Shared ban-check used by every <see cref="IOAuthProvider"/> and by the refresh rotation path (D-03).
    /// Returns <c>null</c> when the player is not banned; returns a <see cref="OAuthResult"/> failure with
    /// error code <c>"banned:&lt;reasonHash&gt;"</c> (first 8 bytes of SHA-256(BanReason) in lowercase hex, 16 chars)
    /// when the player is banned. The reason hash is opaque to the player but lets an admin cross-reference
    /// the audit-log entry containing the full reason.
    /// </summary>
    internal static class BannedCheckHelper
    {
        public static async Task<OAuthResult?> CheckAsync(
            GameKitDbContext ctx, Guid playerId, CancellationToken ct)
        {
            var player = await ctx.Set<Player>().AsNoTracking()
                .FirstOrDefaultAsync(p => p.Id == playerId, ct).ConfigureAwait(false);
            if (player is null || !player.IsBanned) return null;
            var digest = SHA256.HashData(Encoding.UTF8.GetBytes(player.BanReason ?? string.Empty));
            var hex = Convert.ToHexString(digest, 0, 8).ToLowerInvariant(); // 16 chars
            return OAuthResult.Fail($"banned:{hex}");
        }
    }
    ```

        public async Task GuestProvider_BannedPlayer_Login_Returns403_WithReasonHash()
        {
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development");
            var client = host.Client;

            // Login as guest (creates player)
            var loginResp = await client.PostAsJsonAsync("/auth/login/guest", new { device = "test-device" });
            loginResp.EnsureSuccessStatusCode();
            var body = await loginResp.Content.ReadFromJsonAsync<AuthBody>();
            Assert.NotNull(body?.AccessToken);

            // Extract playerId via /auth/me or by decoding JWT
            var playerId = await host.GetPlayerIdByDeviceAsync("test-device");

            // Ban the player via service
            using (var scope = host.Services.CreateScope())
            {
                var bans = scope.ServiceProvider.GetRequiredService<IPlayerBanService>();
                await bans.BanAsync(playerId, actorId: playerId, reason: "test ban", default);

            var resp = await banClient.PostAsJsonAsync("/auth/login/guest", new { device = "test-device-2", playerId });

            var errBody = await resp.Content.ReadFromJsonAsync<ErrBody>();
            Assert.StartsWith("banned:", errBody?.Error ?? "");
            Assert.Matches("^banned:[0-9a-f]{16}$", errBody!.Error);

        public async Task RefreshAfterBan_RevokesFamily_And_Returns401()
        {
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development");
            var client = host.Client;
            var loginResp = await client.PostAsJsonAsync("/auth/login/guest", new { device = "test-device" });
            var body = await loginResp.Content.ReadFromJsonAsync<AuthBody>();
            var playerId = await host.GetPlayerIdByDeviceAsync("test-device");

            using (var scope = host.Services.CreateScope())

                await bans.BanAsync(playerId, actorId: playerId, reason: "test", default);

                Assert.All(families, r => Assert.NotNull(r.RevokedAt));
            }
        }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| banned player → any auth endpoint | Ban is enforced at login (per-provider) + refresh (per-family). Access tokens self-expire; refresh is revoked on first banned-state refresh attempt. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-10-01 | Spoofing | Banned player resumes session via long-lived access token | accept | Access tokens expire within configured TTL (default 15 min). Adversary gets at most 15 min of residual access after ban. RESEARCH documented this tradeoff vs per-request middleware. |
| T-03-10-02 | Information Disclosure | Ban reason leaks to player via response body | mitigate | Only SHA-256 first-8-bytes of BanReason is returned; cannot reverse to plaintext. Admin sees full reason in audit log. |
| T-03-10-03 | Tampering | Attacker replaces ban_reason in Phase 2 DB via SQL injection | mitigate | All player mutations go through EF parameterized queries; no raw SQL in PlayerBanService. |
| T-03-10-04 | Elevation of Privilege | Banned player links a second identity to bypass | mitigate | IdentityLinker enforces `UNIQUE(provider, external_id)`; linking the SAME identity to the banned player still hits the same player row. Linking a DIFFERENT identity creates a separate player (acceptable — "link or switch" policy per AUTH-14). |
| T-03-10-05 | Denial of Service | Attacker repeatedly triggers refresh → family revoke for a target | mitigate | Refresh revoke only affects the banned player's own family; attacker cannot target another player's tokens without possessing their refresh token. |
</threat_model>
