    Step 3 — `MountPathTests.cs`. **SCOPE (revision note, 2026-04-18):** `GameKitAdminOptions.MountPath` relocates the **admin HTTP API prefix only** (`/admin/api/*`). Blazor Server Razor pages are declared with static `@page` routes at `/admin/*` and do **NOT** move with `MountPath`. MudBlazor static assets served at `_content/MudBlazor/*` are root-relative. The test asserts this contract exactly:

    ```csharp
    [Collection("Admin")]
    public class MountPathTests
    {
        private readonly AdminIntegrationFixture _fx;
        public MountPathTests(AdminIntegrationFixture fx) { _fx = fx; }

        [Fact(DisplayName = "MountPath: Custom MountPath relocates API prefix; Blazor shell stays at /admin")]
        public async Task CustomMountPath_RelocatesApiPrefix_And_LeavesBlazorShellAtAdmin()
        {
            // Arrange: AdminTestHost built with AddGameKitAdmin(o => o.MountPath = "/custom-admin-path")
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development",
                configure: builder => builder.AddGameKitAdmin(o => o.MountPath = "/custom-admin-path"),
                seed: h => h.SeedAdminAsync("root", "hunter2hunter2", AdminRoles.Superadmin));

            // Act + Assert (1): API endpoint lives under the custom prefix.
            // Assert NOT 404 (may be 2xx, 401, or 403 depending on anonymous-policy for /health — any of those proves
            // the endpoint is registered under /custom-admin-path/api/*). We reject 404 specifically.
            var customApi = await host.Client.GetAsync("/custom-admin-path/api/health");
            Assert.NotEqual(HttpStatusCode.NotFound, customApi.StatusCode);

            // Act + Assert (2): Default API prefix is gone — /admin/api/* now returns 404.
            var defaultApi = await host.Client.GetAsync("/admin/api/health");
            Assert.Equal(HttpStatusCode.NotFound, defaultApi.StatusCode);

            // Act + Assert (3): Blazor login page is UNCHANGED at /admin/login — MountPath does not move Razor routes.
            var loginPage = await host.Client.GetAsync("/admin/login");
            Assert.True(loginPage.IsSuccessStatusCode,
                $"Expected /admin/login to remain reachable even when MountPath points elsewhere (MountPath scopes API only); got {loginPage.StatusCode}.");
        }
    }
    ```

    [Collection("Admin")]
    public class CspAndAntiforgeryTests
    {
        private readonly AdminIntegrationFixture _fx;
        public CspAndAntiforgeryTests(AdminIntegrationFixture fx) { _fx = fx; }

        [Fact]
        public async Task AdminResponse_Has_ContentSecurityPolicy_Header()
        {
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development",
                seed: h => h.SeedAdminAsync("root", "hunter2hunter2", AdminRoles.Superadmin));

            var resp = await host.Client.GetAsync("/admin/login");
            Assert.True(resp.Headers.TryGetValues("Content-Security-Policy", out var vals));
            var csp = string.Join(";", vals);
            foreach (var d in new[]{"default-src 'self'","frame-ancestors 'none'","script-src 'self'","style-src 'self'","base-uri 'self'","form-action 'self'","img-src 'self' data:"})
                Assert.Contains(d, csp);
        }

        [Fact]
        public async Task TwoSequentialAdminResponses_Have_DifferentNonces()
        {
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development",
                seed: h => h.SeedAdminAsync("root", "hunter2hunter2", AdminRoles.Superadmin));

            var r1 = await host.Client.GetAsync("/admin/login");
            var r2 = await host.Client.GetAsync("/admin/login");
            var n1 = ExtractNonce(r1.Headers.GetValues("Content-Security-Policy").First());
            var n2 = ExtractNonce(r2.Headers.GetValues("Content-Security-Policy").First());
            Assert.NotEqual(n1, n2);
        }

        [Fact]
        public async Task NonAdminResponse_Has_No_CSP_Header()
        {
            // W7: use a PUBLIC player-side endpoint that responds without an auth header (/auth/login/guest
            // accepts anonymous POST). /auth/me was removed from this test because it requires a Bearer
            // token — a 401 response body is irrelevant to the question "does the CSP middleware scope to /admin/*?"
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development",
                seed: h => h.SeedAdminAsync("root", "hunter2hunter2", AdminRoles.Superadmin));

            // Guest login accepts anonymous POST with a device field; request may return 200 or 400 — both are fine
            // for the header-scope assertion. We only care that the response does NOT carry a CSP header, because
            // the AdminCspNonceMiddleware must scope to /admin/* only.
            var resp = await host.Client.PostAsJsonAsync("/auth/login/guest", new { device = "csp-scope-probe" });
            Assert.False(resp.Headers.Contains("Content-Security-Policy"),
                "CSP header must be scoped to /admin/* requests; non-admin paths must not receive it.");
        }

        [Fact]
        public async Task BanMutation_Without_Antiforgery_Returns_400_CsrfValidationFailed()
        {
            await using var host = await AdminTestHost.StartAsync(_fx.Postgres, _fx.Redis, env: "Development",
                seed: h => h.SeedAdminAsync("root", "hunter2hunter2", AdminRoles.Superadmin));
            await host.Client.LoginAsAdminAsync("root", "hunter2hunter2");
            var playerId = await host.SeedPlayerAsync();

            var resp = await host.Client.PostAsJsonAsync($"/admin/api/players/{playerId}/ban", new { reason = "abc" });
            Assert.Equal(HttpStatusCode.BadRequest, resp.StatusCode);
            Assert.Contains("csrf_validation_failed", await resp.Content.ReadAsStringAsync());
        }

        private static string ExtractNonce(string csp)
        {
            var m = System.Text.RegularExpressions.Regex.Match(csp, @"nonce-([A-Za-z0-9+/=]+)");
            return m.Groups[1].Value;
        }
    }
