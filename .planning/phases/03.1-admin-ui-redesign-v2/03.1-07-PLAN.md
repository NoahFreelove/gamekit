<objective>
Ship the Audit page two-column row treatment + the server-side sentence template registry.
The current Audit page renders raw `<pre>` JSON for the `Before` and `After` fields when a
row is expanded — operators read these all day and the JSON is unfriendly. The redesign
replaces this with: a left-column human-readable sentence ("alice banned bob — spam") and a
right-column structured key/value diff with added/removed/changed indicators.

Sentences are computed at READ time (D-13) — the `admin_audit_log` table schema is
unchanged. The `SentenceModel` is projected into the `GET /admin/api/audit` response;
the Razor row template renders from the model. New audit actions whose namespace doesn't
have a registered template fall through to the D-14 generic fallback (`{Actor} performed
{action_words} on {Target} — {reason}`) — never a blank cell.

**Purpose:** D-12 + D-13 lock the registry shape and the read-time projection. UI-SPEC §5.5
codifies the row markup. The 7 currently-known `AdminAuditActions.*` constants get
hand-crafted templates so operator-readable sentences ship from day one.

**Output:** SentenceModel DTO + AuditSentenceTemplates registry + Audit.razor row
template + AdminEndpoints projection extension + live AuditSentenceProjectorTests.
</objective>

```csharp
// SPDX-License-Identifier: GPL-3.0-or-later
// Copyright (c) 2026 GameKit contributors

using System;
using System.Collections.Generic;
using System.Text.Json;
using GameKit.Admin.UI.Http.Contracts;

namespace GameKit.Admin.UI.Services;

/// <summary>
/// Read-time sentence-template registry for the Phase 03.1 audit page (D-12 / D-13). Maps
/// each <see cref="AdminAuditActions"/> namespace constant to a function that builds a
/// <see cref="SentenceModel"/> from the audit row's stored fields. Unknown actions fall
/// through to <see cref="Fallback"/> (D-14) so the UI never shows a blank cell.
/// </summary>
/// <remarks>
/// <para>
/// The registry is computed in-process; no database call. Sentences are recomputed on
/// every <c>GET /admin/api/audit</c> request, which lets template improvements apply
/// retroactively to historical rows without a backfill.
/// </para>
/// </remarks>
public static class AuditSentenceTemplates
{
    private static readonly IReadOnlyDictionary<string, Func<SentenceContext, SentenceModel>> _registry =
        new Dictionary<string, Func<SentenceContext, SentenceModel>>(StringComparer.Ordinal)
        {
            [AdminAuditActions.PlayerBan] = ctx =>
                new SentenceModel(ctx.ActorName, "banned", ctx.TargetName ?? "(unknown player)", null, ctx.Reason),

            [AdminAuditActions.PlayerUnban] = ctx =>
                new SentenceModel(ctx.ActorName, "unbanned", ctx.TargetName ?? "(unknown player)", null, ctx.Reason),

            [AdminAuditActions.PlayerGdprDelete] = ctx =>
                new SentenceModel(ctx.ActorName, "GDPR-deleted", ctx.TargetName ?? "(unknown player)", null, ctx.Reason),

            [AdminAuditActions.PlayerRankAdjust] = ctx =>
                new SentenceModel(ctx.ActorName, "adjusted rank for", ctx.TargetName ?? "(unknown player)",
                    Modifier: ExtractRatingDelta(ctx.Before, ctx.After), Reason: ctx.Reason),

            [AdminAuditActions.AdminCreate] = ctx =>
                new SentenceModel(ctx.ActorName, "created admin", ctx.TargetName ?? "(unknown admin)",
                    Modifier: ExtractRoleAfter(ctx.After), Reason: null),

            [AdminAuditActions.AdminDelete] = ctx =>
                new SentenceModel(ctx.ActorName, "deleted admin", ctx.TargetName ?? "(unknown admin)", null, null),

            [AdminAuditActions.SigningKeyRotate] = ctx =>
                new SentenceModel(ctx.ActorName, "rotated JWT signing key", ctx.TargetName ?? "current key", null, ctx.Reason),
        };

    /// <summary>
    /// Build a sentence model for the given audit row context, falling through to the
    /// generic D-14 fallback when no template matches.
    /// </summary>
    public static SentenceModel Render(SentenceContext ctx)
        => _registry.TryGetValue(ctx.Action, out var fn) ? fn(ctx) : Fallback(ctx);

    /// <summary>
    /// D-14 generic fallback. Operator never sees a blank cell — unmapped actions render
    /// "{Actor} performed {action with dots replaced by spaces} on {Target} — {Reason}".
    /// </summary>
    private static SentenceModel Fallback(SentenceContext ctx)
        => new SentenceModel(
            Actor: ctx.ActorName,
            Intro: "performed",
            Target: ctx.Action.Replace('.', ' '),
            Modifier: ctx.TargetName is null ? null : $"on {ctx.TargetName}",
            Reason: ctx.Reason);

    private static string? ExtractRatingDelta(JsonElement? before, JsonElement? after)
    {
        // Best-effort — when both Before and After contain a "rating" property, render the delta.
        // Returns null on any shape mismatch; the row still renders without the modifier.
        try
        {
            if (before is null || after is null) return null;
            if (before.Value.TryGetProperty("rating", out var b) &&
                after.Value.TryGetProperty("rating", out var a) &&
                b.TryGetDouble(out var bv) && a.TryGetDouble(out var av))
            {
                var delta = av - bv;
                return $"({(delta >= 0 ? "+" : "")}{delta:F1})";
            }
        }
        catch { /* swallow — fallthrough returns null */ }
        return null;
    }

    private static string? ExtractRoleAfter(JsonElement? after)
    {
        try
        {
            if (after is null) return null;
            if (after.Value.TryGetProperty("role", out var r) && r.ValueKind == JsonValueKind.String)
                return $"(as {r.GetString()})";
        }
        catch { /* swallow */ }
        return null;
    }
}

/// <summary>
/// Bundle of the values an audit row contributes to its sentence projection. Built at the
/// projection site in <see cref="GameKit.Admin.UI.Http.AdminEndpoints"/>.
/// </summary>
/// <param name="Action">Stable action namespace from <see cref="AdminAuditActions"/>.</param>
/// <param name="ActorName">Resolved actor display name ("system" for automated).</param>
/// <param name="TargetName">Resolved target display name; null when no target.</param>
/// <param name="Before">Audit row Before JSON (jsonb-backed).</param>
/// <param name="After">Audit row After JSON (jsonb-backed).</param>
/// <param name="Reason">Audit row Reason free-text.</param>
public sealed record SentenceContext(
    string Action,
    string ActorName,
    string? TargetName,
    JsonElement? Before,
    JsonElement? After,
    string? Reason);
```
