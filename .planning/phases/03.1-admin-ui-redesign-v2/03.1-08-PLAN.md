```razor
@* SPDX-License-Identifier: GPL-3.0-or-later *@
@* Copyright (c) 2026 GameKit contributors *@
@*
    UI-SPEC §5.4 ChipRail — filter chip primitive for Audit (action namespace) and Players
    (status). Markup-only state machine: parent passes a list of (Value, Label, IsPressed)
    tuples + an OnToggle callback. Persistence per CONTEXT Claude's Discretion: session-only
    via parent component state; no localStorage.
*@
@namespace GameKit.Admin.UI.Components.Shared

<div class="chip-rail" role="toolbar" aria-label="@Label">
    @foreach (var chip in Chips)
    {
        <button type="button"
                class="chip @(chip.IsPressed ? "chip-active" : "")"
                aria-pressed="@(chip.IsPressed ? "true" : "false")"
                @onclick="() => OnToggle.InvokeAsync(chip.Value)">
            @chip.Label
        </button>
    }
</div>

@code {
    /// <summary>ARIA label for the toolbar (e.g. "Filter by action" / "Filter by status").</summary>
    [Parameter, EditorRequired] public string Label { get; set; } = "Filters";

    /// <summary>Chip data — one entry per filter option. Parent owns selection state.</summary>
    [Parameter, EditorRequired] public IReadOnlyList<ChipItem> Chips { get; set; } = Array.Empty<ChipItem>();

    /// <summary>Invoked when the operator toggles a chip; receives the chip's Value.</summary>
    [Parameter] public EventCallback<string> OnToggle { get; set; }

    /// <summary>Single chip data point — Value is the stable filter key, Label is what the operator sees.</summary>
    /// <param name="Value">Stable filter key passed to OnToggle.</param>
    /// <param name="Label">Operator-facing chip text.</param>
    /// <param name="IsPressed">Whether the chip is currently active (parent-tracked state).</param>
    public sealed record ChipItem(string Value, string Label, bool IsPressed);
}
```
